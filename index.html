from abc import ABC, abstractmethod class Student(ABC): university = "Istanbul Gelisim Universitesi" # Class Variable student_count = 0 def __init__(self, name, student_id, gpa): self.__name = name # private — Encapsulation self.__student_id = student_id self.__gpa = 0.0 self.set_gpa(gpa) Student.student_count += 1 def get_name(self): return self.__name def get_student_id(self): return self.__student_id def get_gpa(self): return self.__gpa def set_gpa(self, gpa): if 0.0 <= gpa <= 4.0: self.__gpa = gpa else: raise ValueError("GPA must be between 0.0 and 4.0") @property def info(self): return f"{self.__name} | {self.__student_id}" def __str__(self): return f"Student({self.__name}, GPA: {self.__gpa})" @abstractmethod def study(self): pass @abstractmethod def calculate_fee(self): pass
class UnderGrad(Student): # Inheritance def __init__(self, name, student_id, major, gpa): super().__init__(name, student_id, gpa) # super() self.__major = major def get_major(self): return self.__major def study(self): # Override + Abstraction return "Reading textbooks and attending lectures" def calculate_fee(self): # Override + Polymorphism return 3500 def __str__(self): return f"UnderGrad({self.get_name()}, Major: {self.__major})" yahya = UnderGrad("Yahya", "250418770", "AI & Data Science", 3.85) abdelhady = UnderGrad("Abdelhady", "250418771", "Artificial Intelligence", 3.70) layla = UnderGrad("Layla", "250418772", "Software Engineering", 3.92)
class PostGrad(Student): # Inheritance def __init__(self, name, student_id, research_topic, gpa): super().__init__(name, student_id, gpa) self.__research_topic = research_topic def get_research_topic(self): return self.__research_topic def study(self): return "Writing research papers and conducting experiments" def calculate_fee(self): return 6500 def __str__(self): return f"PostGrad({self.get_name()}, Research: {self.__research_topic})"
# Multiple Inheritance — Python exclusive! class Researcher: def conduct_research(self): return "Conducting AI/ML research..." class AIStudent(UnderGrad, Researcher): # inherits from BOTH def __init__(self, name, student_id, major, gpa): super().__init__(name, student_id, major, gpa) yahya = AIStudent("Yahya", "250418770", "AI & Data Science", 3.85) print(isinstance(yahya, AIStudent)) # True print(isinstance(yahya, UnderGrad)) # True print(isinstance(yahya, Researcher)) # True print(isinstance(yahya, Student)) # True — full chain!
